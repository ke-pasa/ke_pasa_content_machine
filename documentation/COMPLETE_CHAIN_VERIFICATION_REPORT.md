```python
def run_hourly_publication(self):
    # 1. Проверяет время публикации
    publication_hours = [9, 10, 11, 14, 15, 16, 20, 21]
    if current_hour not in publication_hours:
        return
    
    # 2. Получает неопубликованные статьи
    articles = self._get_fresh_unpublished_articles()
    
    # 3. Ранжирует статьи по рейтингу
    ranked_articles = self._rank_articles_for_telegram(articles, settings)
    
    # 4. Выбирает 1 лучшую статью
    best_article = ranked_articles[0]
    
    # 5. Генерирует Telegram-пост
    telegram_post = telegram_generator.generate_post(best_article, article_url)
    
    # 6. Публикует в Telegram
    if self._send_telegram_post(best_article, settings):
        # 7. Отмечает как опубликованную
        self._mark_article_as_published(best_article)
```

```python
def rank_for_telegram(openai_client, firebase_client, articles, settings):
    for article in articles:
        # LLM-оценка
        assess = llm_quick_assess(openai_client, article)

        score = simple_score(article, assess)
```
